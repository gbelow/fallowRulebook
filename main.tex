% --- Reusable content patterns --------------------------------------------

% stattable{title}{colspec}: a gear/stat table with a caption above and
% booktabs rules. Shade an item's row(s) with \rowcolor{fallowtint} to mark
% every other entry — do this per item, not per raw row, since many entries
% span two rows (e.g. a melee profile and a thrown profile).
\newenvironment{stattable}[2]{%
  \begin{table}[H]
  \caption{#1}
  \centering
  \footnotesize
  \renewcommand{\arraystretch}{1.3}
  \begin{tabular}{#2}
  \toprule
}{%
  \bottomrule
  \end{tabular}
  \end{table}
}

% proplist: a glossary-style list (\item \textbf{Term:} definition...)
% with tightened spacing and an accent-colored marker instead of a bullet.
\newenvironment{proplist}{%
  \begin{itemize}[
    label={\textcolor{fallowrule}{\textbullet}},
    leftmargin=1.1em,
    itemsep=0.15em,
    parsep=0pt,
    topsep=0.3em
  ]
}{%
  \end{itemize}
}

% dnote: a Designer's Note callout. The framed package cannot break a box
% across multicols columns, so — like stattable — it closes the two-column
% flow and sits at full width rather than squeezed into one column.
\newenvironment{dnote}{%
  \def\FrameCommand{\fboxsep=7pt\fboxrule=0.6pt\fcolorbox{fallowrule}{fallowtint}}%
  \MakeFramed{\advance\hsize-\width\FrameRestore}%
  \noindent{\sffamily\bfseries\footnotesize\textcolor{fallowrule}{Designer's Note}}\par\vspace{2pt}%
  \normalfont\itshape\small
}{%
  \par\endMakeFramed
}

% dnotesm: a smaller Designer's Note that stays inside one column of a
% multicols layout (dnote cannot — see above). Built as an ordinary box,
% not a framed one, since it never needs to break across a page/column:
% keep its content short enough to fit in the remaining column height.
\newsavebox{\dnotesmbox}
\newenvironment{dnotesm}{%
  \begin{lrbox}{\dnotesmbox}
  \begin{minipage}{0.94\linewidth}
  \noindent{\sffamily\bfseries\scriptsize\textcolor{fallowrule}{Designer's Note}}\par\vspace{1pt}%
  \normalfont\itshape\footnotesize
}{%
  \end{minipage}
  \end{lrbox}
  \par\noindent\fboxsep=5pt\fboxrule=0.5pt\fcolorbox{fallowrule}{fallowtint}{\usebox{\dnotesmbox}}\par
}
